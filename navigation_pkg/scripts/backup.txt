#!/usr/bin/env python

import rospy
import math
from sensor_msgs.msg import LaserScan
from nav_msgs.msg import Odometry
from gazebo_msgs.msg import ModelStates
from geometry_msgs.msg import Twist
import numpy as np
import transforms3d.euler as t3d_euler
from navigation_utils import calcDanger, calc_h, calc_hp, calc_Hb, find_valleys, calc_Target, pick_valley, pick_heading


class NavigationNode:
    def __init__(self):
        rospy.init_node('navigation_node', anonymous=True)
        self.cmd_vel_pub = rospy.Publisher('/cmd_vel', Twist, queue_size=10)
        self.lidar_sub = rospy.Subscriber('/front/scan', LaserScan, self.lidar_callback)
        self.odom_sub = rospy.Subscriber('/odometry/filtered', Odometry, self.odom_callback)
        self.model_sub = rospy.Subscriber('/gazebo/model_states', ModelStates, self.model_callback)

        self.processed_lidar_ranges = None
        self.odom_data = None
        self.model_data = None
        self.max_range = 4.0       # Maximum range for danger calculation
        self.sector_size = 8         # Group size for reducing Lidar data resolution
        self.filter_width = 3        # Filter width (L) for smoothing
        self.threshold = 1.0         # Threshold (in MATLAB, threshold is set to 1)
        self.robotDim = 0.5
        self.current_position = (0, 0)
        self.current_heading = 0
        self.WidevalleyMin = 18
        

        # Constant start and target positions

        self.INIT_POSITION = [-2, 3, 1.57]  # in world frame
        self.GOAL_POSITION = [0, 10]  # relative to the initial position
        self.target_absolute_position = (self.INIT_POSITION[0] + self.GOAL_POSITION[0], self.INIT_POSITION[1] + self.GOAL_POSITION[1])


    def lidar_callback(self, msg):
         # Extract raw lidar ranges from the message
        lidar_range = np.array(msg.ranges)
        
        # Reverse the order of the ranges (like MATLAB's flipud)
        lidar_ranges = np.flip(lidar_range)
        
        # Limit values to max_range (self.max_range is defined in your node)
        lidar_ranges[lidar_ranges > self.max_range] = self.max_range
        
        # Optionally store the processed ranges in a class variable for later use
        self.processed_lidar_ranges = lidar_ranges

        #rospy.loginfo("Processed lidar ranges: %s", lidar_ranges)
           

    def odom_callback(self, msg):
        self.odom_data = msg
         #rospy.loginfo("Odometry data: %s", self.odom_data)

         # Extract the robot's position from the odometry message.
        x = msg.pose.pose.position.x
        y = msg.pose.pose.position.y
        z = msg.pose.pose.position.z

        # print("x: ", x)
        # print("y: ", y)
        # Adjust the robot's position by adding the starting offset.
        current_x = x + self.INIT_POSITION[0]
        current_y = y + self.INIT_POSITION[1]
        # print("current x: ", current_x)
        # print("current y: ", current_y)
        
        # Convert the quaternion to Euler angles using transforms3d.
        # ROS quaternion is given as (x, y, z, w). However, transforms3d expects [w, x, y, z]
        orientation_q = msg.pose.pose.orientation
        q_ros = [orientation_q.w, orientation_q.x, orientation_q.y, orientation_q.z]
        # Convert quaternion to Euler angles (roll, pitch, yaw) in radians.
        roll, pitch, yaw = t3d_euler.quat2euler(q_ros, axes='sxyz')
        # Convert yaw from radians to degrees.
        theta = math.degrees(yaw)
       
        
        # Normalize theta to the range [-180, 180] if needed.
        if theta < -180:
            theta += 360
        elif theta > 180:
            theta -= 360
        

        # Adjust current heading by adding an offset. 
        # Do we need to add a starting value to this angle??? like the MATLAB TODO
        currHeading = theta #+ 90

        # Store the computed current position and heading.
        self.current_position = (current_x, current_y)
        self.current_heading = currHeading

        #rospy.loginfo("Odometry processed: current_position = (%.2f, %.2f), current_heading = %.2f", current_x, current_y, currHeading)




    def model_callback(self, msg):
        self.model_data = msg
        #rospy.loginfo("Model states data received.")

    def run(self):
        rate = rospy.Rate(10)  # 10 Hz
        while not rospy.is_shutdown():
            # Example usage of calcDanger:
            if self.processed_lidar_ranges is not None:
                
                m = calcDanger(self.processed_lidar_ranges, self.max_range)
                #rospy.loginfo("Calculated danger vector (m): %s", str(m))

                h = calc_h(m, self.sector_size)
                #rospy.loginfo("Reduced Polar vector (h): %s", h)

                hp = calc_hp(h, self.filter_width)
                #rospy.loginfo("Smoothed vector (hp): %s", hp)

                hb = calc_Hb(hp, self.threshold)
                #rospy.loginfo("Computed hb = %s", hb)



                valleys, sectToClear = find_valleys(hb, self.threshold, self.robotDim)
                rospy.loginfo("passable valleys (start, end)= %s", valleys)
                #rospy.loginfo("Sectors Required to clear= %s", sectToClear)

                # print("target_absolute_position: ", self.target_absolute_position)
                # print("self.current_position: ", self.current_position)
                # print("self.current_heading: ", self.current_heading)
                heading_sector = calc_Target(self.target_absolute_position, self.current_position, self.current_heading)
                #rospy.loginfo("target sector: %d", heading_sector)


                chosen_valley = pick_valley(valleys, heading_sector)
                #rospy.loginfo("Chosen valley: %s", chosen_valley)

                heading_final = pick_heading(heading_sector, chosen_valley, self.WidevalleyMin, sectToClear)
                rospy.loginfo("final heading: %d", heading_final)





            # Publish a zero velocity command as a placeholder
            twist = Twist()
            twist.linear.x = 0.0
            twist.angular.z = 0.0
            self.cmd_vel_pub.publish(twist)
            rate.sleep()

if __name__ == '__main__':
    try:
        node = NavigationNode()
        node.run()
    except rospy.ROSInterruptException:
        pass
